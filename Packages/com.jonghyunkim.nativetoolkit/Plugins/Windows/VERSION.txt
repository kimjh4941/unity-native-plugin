# Which native-toolkit artifacts this folder holds, and where they came from.
#
# PreBuildProcessor READS the keys below from this file. It does not scan dist
# and guess. If a required key is missing, or the artifact it names is not in
# dist, the build FAILS rather than copying something else.
#
# Change these keys in the same commit as the DLL. The whole point is that the
# binary in this folder and the line that names it move together.
#
# dist_version      the folder under native-toolkit/dist to copy from (required)
# release_dll       the artifact to use for release builds (required)
# debug_dll         the artifact for development builds; empty = use release_dll
#                   (dist has never published a -debug.dll for Windows)
# install_as        the file name to place here; empty = keep the dist file name
# install_as_debug  the file name for development builds; empty = install_as,
#                   then the dist file name
#
# extra_*           a second pin, placed beside the main one. Only while the
#                   Windows features move to the 2.0.0 C ABI one at a time: the
#                   features not moved yet still load the 1.x DLL. The keys mean
#                   what the main ones do (extra_dist_version, extra_dll,
#                   extra_install_as, extra_install_as_debug); there is no
#                   extra_debug_dll, so the source is always extra_dll. Remove
#                   them, and the 1.x DLL, once Clipboard has moved (the last one).
#                   artifact/features/dialog/designs/2026-09-27-windows-dialog-design-v6.md, J-6
#
# Each name placed here must match the DLL name its managers load:
# - main pin (2.0.0 C ABI): WindowsNativeToolkitCApi.DllName in
#   Runtime/Windows/Common, with ".dll". The name contains '.', so it is written
#   out in full rather than left for the loader to complete.
# - extra pin (1.x): DLL_NAME in the managers still on 1.x, without ".dll".
#   Dialog and Notification have moved; only Runtime/Windows/Clipboard loads it now.

# Main pin: the 2.0.0 C ABI, under its own dist name for both builds.
dist_version: 1.12.0
release_dll: windows-native-toolkit-capi-2.0.0.dll
debug_dll:
install_as:
install_as_debug:
source: https://github.com/kimjh4941/native-toolkit/tree/feature/NTKIT-16/dist/1.12.0/windows/windows-native-toolkit-capi-2.0.0.dll

# Extra pin: the 1.x DLL. The native library's own version is 1.2.0, which is
# not the same number as dist_version. The 1.x P/Invoke layer loads it as
# "unity-windows-native-toolkit" (release) or "unity-windows-native-toolkit-debug"
# (DEVELOPMENT_BUILD), hence the two install_as names.
extra_dist_version: 1.11.0
